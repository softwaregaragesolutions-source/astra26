Hardware Wiring Schematic (5 Sensors)

1
DS18B20 Temp Pin → Connect Data (Yellow) to GPIO 4 + 4.7kΩ resistor to VCC.
2
LDR Light Sensor Pin → Connect Analog Out (AO) to ESP32 GPIO 34.
3
Voltage Divider Module Pin → Connect Signal Out to ESP32 GPIO 35.
4
Raindrop Sensor Module Pin → Connect Analog Out (AO) to ESP32 GPIO 32.
5
Tilt Switch Sensor Pin → Connect Digital Out (DO) to ESP32 GPIO 25.

Render Hosting Steps

  1. Push this repository to GitHub.
  2. Go to Render.com and sign in.
  3. Click New + → Select Web Service.
  4. Connect your GitHub repo.
  5. Set Environment to Node.
  6. Build Command: npm install
  7. Start Command: npm start
  8. Click Create Web Service! Your API URL will be: https://your-app-name.onrender.com/api/temperature

Install Arduino Libraries: OneWire & DallasTemperature before compiling.

 * ESP32 + DS18B20 Waterproof Temperature Sensor (HTTP & HTTPS Cloud Compatible)
 * Automatically sends data to your Render cloud endpoint
 */

#include

#include 
#include 

const char* WIFI_SSID     = "YOUR_WIFI_SSID";
const char* WIFI_PASSWORD = "YOUR_WIFI_PASSWORD";
const char* SERVER_URL    = "https://astra26.onrender.com/api/temperature";

#define ONE_WIRE_BUS 4
#define STATUS_LED 2

OneWire oneWire(ONE_WIRE_BUS);
DallasTemperature sensors(&oneWire);

void setup() {
  Serial.begin(115200);
  pinMode(STATUS_LED, OUTPUT);
  sensors.begin();
  
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nConnected to WiFi!");
}

void loop() {
  sensors.requestTemperatures();
  float tempC = sensors.getTempCByIndex(0);

  if (tempC != DEVICE_DISCONNECTED_C && WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    WiFiClientSecure client;
    if (String(SERVER_URL).startsWith("https")) {
      client.setInsecure();
      http.begin(client, SERVER_URL);
    } else {
      http.begin(SERVER_URL);
    }
    http.addHeader("Content-Type", "application/json");

    String json = "{\"temperature\":" + String(tempC, 2) + ",\"sensor_id\":\"DS18B20_ESP32\"}";
    int httpCode = http.POST(json);
    Serial.printf("POST Code: %d\n", httpCode);
    http.end();
  }
  delay(10000); // 10s interval
}
